Sabiendo x,y de la diagonal, ¿Cómo se cuantos valores de diagonal tengo que imprimir? el maximo de x,y menos len(M) no da ese valor

def imprimir_diagonal(M,x,y):
    diagonal=[]
    long = len(M)-max(x,y)
    for i in range(len(M)):
        if long!=0:
            diagonal.append(M[x+i][y+i])
            long-=1
    return diagonal

print(imprimir_diagonal([[1,2,3,9],[4,5,6,4],[7,8,9,5],[2,3,4,5]],0,2))
